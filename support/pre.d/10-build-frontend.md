# Build frontend procedure (EXAMPLE)

This file is an EXAMPLE of a pre-build procedure that can be added to the support/pre.d directory. Do not edit this file directly. Read [README.md](README.md) for more information.

## Code

```bash
#!/usr/bin/env bash
# shellcheck disable=SC2065,SC2015

set -o nounset           # Treat unset variables as an error
set -o errexit           # Exit immediately if a command exits with a non-zero status
set -o pipefail          # Prevent errors in a pipeline from being masked
set -o errtrace          # If a command fails, the shell will exit immediately
set -o functrace         # If a function fails, the shell will exit immediately
shopt -s inherit_errexit # Inherit the errexit option in functions

IFS=$'\n\t'

_ROOT_DIR=$(realpath "$(git rev-parse --show-toplevel)")
_FRONTEND_DIR=$(realpath "${FRONTEND_DIR:-"${_ROOT_DIR}/frontend"}")
_INTERNAL_FRONTEND_DIR=$(realpath "${INTERNAL_FRONTEND_DIR:-"${_ROOT_DIR}/internal/features/ui/web"}")

__source_script_if_needed() {
  local _check_declare="${1:-}"
  local _script_path="${2:-}"
  # shellcheck disable=SC2065
  if test -z "$(declare -f "${_check_declare:-}")" >/dev/null; then
    # shellcheck source=/dev/null
    source "${_script_path:-}" || {
      echo "Error: Could not source ${_script_path:-}. Please ensure it exists." >&2
      return 1
    }
  fi
  return 0
}

_SCRIPT_DIR=$(realpath "$(cd "$(dirname "${0}")" && pwd || exit 1)")
if git rev-parse --show-toplevel >/dev/null 2>&1; then
  _SCRIPT_DIR=$(realpath "$(git rev-parse --show-toplevel)")
fi

__source_script_if_needed "get_current_shell" "${_SCRIPT_DIR:-}/utils.sh" || {
  echo "Error: Could not source utils.sh. Please ensure it exists." >&2
  exit 1
}

build_frontend() {
  if [[ ! -d ${_FRONTEND_DIR} ]]; then
    echo "Frontend directory does not exist."
    exit 1
  fi

  cd "${_FRONTEND_DIR}" || {
    log fatal "Failed to change directory to ${_FRONTEND_DIR}" || echo "Failed to change directory to ${_FRONTEND_DIR}" >&2
    exit 1
  }

  if command -v pnpm &>/dev/null; then
    log notice "Building frontend..."

    _frontend_install_output="$(pnpm install || {
      echo "Failed to install frontend dependencies."
    })"

    _frontend_build_output="$(pnpm build >/dev/null 2>&1 || {
      echo "Failed to build frontend assets."
    })"

    if [[ ${_frontend_build_output:-} == "Failed to build frontend assets." ]] || [[ -n ${_frontend_build_output:-} && ${_QUIET:-false} != "true" ]]; then
      log error "${_frontend_build_output}"
      log fatal "Frontend build failed."
      exit 1
    fi

    if [[ ! -d "${_FRONTEND_DIR}/dist" ]]; then
      log fatal "Build directory does not exist."
      exit 1
    fi

    if [[ -d ${_INTERNAL_FRONTEND_DIR} ]]; then
      log notice "Removing old build directory..."
      rm -rf "${_INTERNAL_FRONTEND_DIR}" || {
        log fatal "Failed to remove old build directory."
        exit 1
      }
    fi

    if [[ ${KEEP_DIST_DIR:-true} != "true" || ${BUILD_MODE:-dev} != "dev" ]]; then
      mv -f './dist' "${_INTERNAL_FRONTEND_DIR}" || {
        log fatal "Failed to move build directory to internal/features/ui/web."
        exit 1
      }
      log success "Frontend build moved to internal/features/ui/web directory successfully."
    else
      cp -r './dist' "${_INTERNAL_FRONTEND_DIR}" || {
        log fatal "Failed to copy build directory to internal/features/ui/web."
        exit 1
      }
      log success "Frontend build copied to internal/features/ui/web directory successfully."
    fi
  else
    log fatal "npm is not installed. Please install Node.js and npm to continue."
    exit 1
  fi
}

# Executa o hook pré build: valida deps; builda o FE; move pro embedded go;
if [[ ${_RUN_PRE_SCRIPTS:-${RUN_PRE_SCRIPTS:-true}} == "true" ]]; then
  (build_frontend) || {
    log fatal "An error occurred during the pre-build process." || echo "An error occurred during the pre-build process." >&2
    exit 1
  }
fi

```
